      \begin{itemize}
        \item<1-> Maintain an array \texttt{parent[]}
        \item<2-> \texttt{parent[i]} stores the index of element $i$'s \textbf{immediate parent} --- not necessarily its representative.
        \item<3-> Following parents eventually reaches the \textbf{root}, which is exactly the representative: a root points to itself, \texttt{parent[root] = root}.
      \end{itemize}
    \end{column}
